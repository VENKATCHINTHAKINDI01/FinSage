# FinSage AI — runtime dependencies
#
# Dev/test tooling lives in requirements-dev.txt.
#
# Pinning policy: compatible-release (~=) on the minor for libraries that
# respect semver, exact pins only where a break is known. v1 pinned everything
# exactly and then never moved, which is how it ended up ~2 years behind on
# FastAPI, Starlette and Pydantic.
#
# Audited 2026-08-09 (FND-005).

# ── Web framework ────────────────────────────────────────────────────────────
fastapi~=0.141.1
uvicorn[standard]~=0.52.1
python-multipart~=0.0.32
# starlette is deliberately NOT pinned here — fastapi constrains it, and
# pinning both is how you get an unsolvable resolution.

# ── Config & validation ──────────────────────────────────────────────────────
pydantic~=2.13.4
pydantic-settings~=2.7
email-validator~=2.2

# ── Database ─────────────────────────────────────────────────────────────────
sqlalchemy[asyncio]~=2.0.51

redis~=8.1.0
# psycopg2-binary removed: asyncpg is the only driver in use, and alembic is

langgraph~=0.6
# langchain / langchain-community / langchain-groq / langchain-text-splitters
# removed: nothing imports them. Only langgraph is referenced

# ── Vector search & embeddings ───────────────────────────────────────────────
qdrant-client~=1.19.0
fastembed~=0.7.4
# Replaces sentence-transformers, which v1 pinned at 2.2.2 (broken against
# modern huggingface_hub) and never actually used — embeddings are currently
# md5 -> np.random.seed -> random vector.
#
# fastembed is Qdrant's own ONNX-runtime embedding library. It serves the same
# BAAI/bge-small-en-v1.5 model but pulls no torch, no transformers and no CUDA
# toolkit: measured against this exact requirement set, sentence-transformers
# adds torch 2.13 plus ~18 nvidia-* packages (multiple GB) to the image for a
# CPU-only workload. Real embeddings land in DEM-004.
numpy~=2.1

# ── HTTP ─────────────────────────────────────────────────────────────────────
httpx~=0.28.1
# tavily-python removed: web_search.py calls the Tavily REST API through httpx
# directly and never imported the SDK.
# aiohttp removed: nothing imports it.

# ── Security ─────────────────────────────────────────────────────────────────
PyJWT~=2.13.0
bcrypt~=5.0.0
passlib[bcrypt]~=1.7.4
# TODO(PRD-002): passlib is effectively unmaintained and its bcrypt backend
# warns against bcrypt>=4.1. Replace with a direct bcrypt call in
# backend/security/password.py — existing $2b$ hashes stay valid, so this is a
# drop-in change, not a migration.
# python-jose removed: unused, unmaintained, and PyJWT already covers it.
cryptography~=46.0
